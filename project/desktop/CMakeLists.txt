cmake_minimum_required(VERSION 3.22)
project(RhythmerDesktop VERSION 0.1.0 LANGUAGES C CXX OBJC OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(JUCE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../juce" CACHE PATH "JUCE source")
add_subdirectory("${JUCE_PATH}" juce EXCLUDE_FROM_ALL)
juce_add_plugin(RhythmerDesktop COMPANY_NAME "Jazzy Rhythmer" BUNDLE_ID "com.jazzyrhythmer.rhythmer-engines" PLUGIN_MANUFACTURER_CODE JzRh PLUGIN_CODE Rhen FORMATS AU VST3 PRODUCT_NAME "Rhythmer Engines" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE VST3_AUTO_MANIFEST FALSE AU_MAIN_TYPE kAudioUnitType_MusicDevice)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "../src/animation/performance_art.inc" "../tools/prepare_desktop_art.py")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../tools/prepare_desktop_art.py" "${CMAKE_BINARY_DIR}/generated/desktop_art_context.inc" COMMAND_ERROR_IS_FATAL ANY)
target_sources(RhythmerDesktop PRIVATE Desktop.cpp LiveArt.cpp
 "../native/Rhythmer Engines/src/rhythmer/model/pattern.cpp"
 "../native/Rhythmer Engines/src/rhythmer/model/gen8.cpp"
 "../native/Rhythmer Engines/src/rhythmer/model/chord_gen.cpp")
target_include_directories(RhythmerDesktop PRIVATE "${CMAKE_BINARY_DIR}/generated")
target_include_directories(RhythmerDesktop PRIVATE "../native/Rhythmer Engines/src" "../native/Rhythmer Engines/src/framework")
target_compile_definitions(RhythmerDesktop PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(RhythmerDesktop PRIVATE juce::juce_audio_utils juce::juce_gui_extra PUBLIC juce::juce_recommended_config_flags)
add_library(RhythmerVST2 MODULE VST2Shell.mm)
target_link_libraries(RhythmerVST2 PRIVATE RhythmerDesktop)
target_include_directories(RhythmerVST2 PRIVATE "${JUCE_PATH}/modules" "../native/Rhythmer Engines/src/framework")
target_compile_definitions(RhythmerVST2 PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
set_target_properties(RhythmerVST2 PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION vst PREFIX "" OUTPUT_NAME "Rhythmer Engines" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/VST2Info.plist" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/RhythmerDesktop_artefacts/Release/VST2")
# Keep framework implementation symbols inside each plug-in. In particular,
# coalescing JUCE's weak VariantType tables with a host/another format can route
# String cleanup to the wrong image's static empty-string storage.
target_link_options(RhythmerVST2 PRIVATE "LINKER:-exported_symbol,_VSTPluginMain")
target_link_options(RhythmerDesktop_AU PRIVATE "LINKER:-exported_symbol,_Rhythmer_EnginesAUFactory" "LINKER:-exported_symbol,_JuceAUFactory")
target_link_options(RhythmerDesktop_VST3 PRIVATE "LINKER:-exported_symbol,_GetPluginFactory" "LINKER:-exported_symbol,_bundleEntry" "LINKER:-exported_symbol,_bundleExit")

option(RHYTHMER_BUILD_QA_HOST "Build the local AU/VST3 audio and editor test host" OFF)
if(RHYTHMER_BUILD_QA_HOST)
 juce_add_console_app(RhythmerMacQA PRODUCT_NAME "Rhythmer Mac QA")
 target_sources(RhythmerMacQA PRIVATE ../tests/mac_formats.mm)
 target_include_directories(RhythmerMacQA PRIVATE "../native/Rhythmer Engines/src/framework")
 target_compile_definitions(RhythmerMacQA PRIVATE JUCE_MODAL_LOOPS_PERMITTED=1 JUCE_PLUGINHOST_AU=1 JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
 target_link_libraries(RhythmerMacQA PRIVATE juce::juce_audio_utils juce::juce_gui_extra)
 add_executable(RhythmerArtQA ../tests/live_art.cpp LiveArt.cpp)
 target_include_directories(RhythmerArtQA PRIVATE "${CMAKE_BINARY_DIR}/generated" "${CMAKE_CURRENT_SOURCE_DIR}")
 juce_add_console_app(RhythmerEditorQA PRODUCT_NAME "Rhythmer Editor QA")
 target_sources(RhythmerEditorQA PRIVATE ../tests/editor_layout.cpp)
 target_include_directories(RhythmerEditorQA PRIVATE "${JUCE_PATH}/modules")
 target_compile_definitions(RhythmerEditorQA PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_MODAL_LOOPS_PERMITTED=1)
 target_link_libraries(RhythmerEditorQA PRIVATE RhythmerDesktop)
endif()
option(RHYTHMER_QA_ASAN "Instrument the desktop adapter and QA host" OFF)
if(RHYTHMER_QA_ASAN)
 foreach(t RhythmerDesktop RhythmerMacQA)
  target_compile_options(${t} PRIVATE -fsanitize=address -fno-omit-frame-pointer)
 endforeach()
 foreach(t RhythmerDesktop_AU RhythmerDesktop_VST3 RhythmerVST2 RhythmerMacQA)
  target_link_options(${t} PRIVATE -fsanitize=address)
 endforeach()
endif()
if(RHYTHMER_BUILD_QA_HOST)
 juce_add_console_app(RhythmerMidiQA PRODUCT_NAME "Rhythmer MIDI QA")
 target_sources(RhythmerMidiQA PRIVATE ../tests/mac_midi_workspace.cpp)
 target_include_directories(RhythmerMidiQA PRIVATE "${JUCE_PATH}/modules" "../native/Rhythmer Engines/src" "../native/Rhythmer Engines/src/framework")
 target_link_libraries(RhythmerMidiQA PRIVATE RhythmerDesktop)
endif()
